cmake_minimum_required(VERSION 3.20)
project(motion_guard LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(motion_guard INTERFACE)
target_include_directories(motion_guard INTERFACE include)
foreach(target IN ITEMS experiment scenario_eval)
  add_executable(${target} src/${target}.cpp)
  target_link_libraries(${target} PRIVATE motion_guard)
endforeach()
add_executable(geometry_tests tests/test.cpp)
target_link_libraries(geometry_tests PRIVATE motion_guard)
enable_testing()
add_test(NAME geometry COMMAND geometry_tests)
add_test(NAME scenario COMMAND experiment --scenario)
